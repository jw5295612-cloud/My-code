! =====================================================================
! CAFE-SW1  -  Cisco 2960-24TT access switch
! Paste in:  Switch> enable  ->  Switch# configure terminal
! =====================================================================
hostname CAFE-SW1
no ip domain-lookup
service password-encryption
enable secret Cafe@Enable1
banner motd #Authorized access only - CyberCafe LAN#
ip domain-name cybercafe.local
username admin privilege 15 secret Cafe@Admin1
!
vlan 10
 name CUSTOMER-PCS
vlan 20
 name PRINTERS
vlan 30
 name CCTV
vlan 99
 name MANAGEMENT
vlan 999
 name NATIVE-UNUSED
!
! ---------- Customer PCs (PC1 = admin/cashier, PC2-PC4 = customers) ----------
interface range FastEthernet0/1 - 4
 description CUSTOMER-PC
 switchport mode access
 switchport access vlan 10
 switchport nonegotiate
 switchport port-security
 switchport port-security maximum 1
 switchport port-security mac-address sticky
 switchport port-security violation restrict
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown
!
! ---------- Printers ----------
interface range FastEthernet0/5 - 6
 description PRINTER
 switchport mode access
 switchport access vlan 20
 switchport nonegotiate
 switchport port-security
 switchport port-security maximum 1
 switchport port-security mac-address sticky
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown
!
! ---------- CCTV IP cameras ----------
interface range FastEthernet0/7 - 8
 description CCTV-CAMERA
 switchport mode access
 switchport access vlan 30
 switchport nonegotiate
 switchport port-security
 switchport port-security maximum 1
 switchport port-security mac-address sticky
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown
!
! ---------- Unused ports: parked and shut ----------
interface range FastEthernet0/9 - 24 , GigabitEthernet0/2
 description UNUSED
 switchport mode access
 switchport access vlan 999
 shutdown
!
! ---------- Uplink trunk to CAFE-R1 ----------
interface GigabitEthernet0/1
 description TRUNK to CAFE-R1 G0/0
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,30,99,999
 switchport nonegotiate
 no shutdown
!
! ---------- DHCP snooping: only the uplink may hand out leases ----------
ip dhcp snooping
ip dhcp snooping vlan 10
no ip dhcp snooping information option
interface GigabitEthernet0/1
 ip dhcp snooping trust
!
! ---------- Management SVI ----------
interface Vlan1
 shutdown
interface Vlan99
 description MANAGEMENT
 ip address 192.168.99.2 255.255.255.0
 no shutdown
ip default-gateway 192.168.99.1
!
crypto key generate rsa general-keys modulus 1024
ip ssh version 2
line console 0
 login local
line vty 0 15
 login local
 transport input ssh
!
end
write memory
